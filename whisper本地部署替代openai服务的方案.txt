    - 句子翻译: 中英文翻译和润色

  // 语法分析

  2. 阿里云ASR服务实现

  export class AliyunWhisperRequest implements Cancelable {
      private abortController: AbortController | null = null;

      @WaitRateLimit('aliyun-asr')
      public async invoke(): Promise<WhisperResponse> {
          this.abortController = new AbortController();

          try {
              // 调用阿里云ASR API
              const response = await
  this.aliyunClient.request('RecognizeAudio', {
                  audio: fs.createReadStream(this.file),
                  format: 'wav',
                  sample_rate: 16000,
                  enable_sample_rate_adaptive: true,
                  enable_words: true, // 获取词级别时间戳
                  enable_timestamp: true,
                  enable_punctuation: true,
                  enable_inverse_text_normalization: true,
              });

              // 转换为标准WhisperResponse格式
              return this.convertToWhisperFormat(response);
          } catch (error) {
              dpLog.error('阿里云ASR调用失败:', error);
              throw error;
          }
      }

      private convertToWhisperFormat(aliyunResponse: any): WhisperResponse {
          return {
              language: 'zh',
              duration: aliyunResponse.duration,
              text: aliyunResponse.text,
              segments: aliyunResponse.result.sentences?.map((sentence: any)
   => ({
                  seek: 0,
                  start: sentence.begin_time / 1000, // 毫秒转秒
                  end: sentence.end_time / 1000,
                  text: sentence.text
              })) || []
          };
      }
  }

  创建本地Whisper服务:

  // src/backend/services/impl/LocalWhisperService.ts
  import { spawn, ChildProcess } from 'child_process';
  import path from 'path';

  export class LocalWhisperService {
      private pythonProcess: ChildProcess | null = null;

      public async transcribe(filePath: string, model = 'medium.en'):
  Promise<WhisperResponse> {
          return new Promise((resolve, reject) => {
              const script = path.join(__dirname, 'whisper_runner.py');

              this.pythonProcess = spawn('python3', [script, filePath,
  '--model', model]);

              let output = '';
              this.pythonProcess.stdout?.on('data', (data) => {
                  output += data.toString();
              });

              this.pythonProcess.on('close', (code) => {
                  if (code === 0) {
                      const result = JSON.parse(output);
                      resolve(this.convertToWhisperFormat(result));
                  } else {
                      reject(new Error('Whisper执行失败'));
                  }
              });
          });
      }

      public cancel(): void {
          if (this.pythonProcess) {
              this.pythonProcess.kill();
              this.pythonProcess = null;
          }
      }
  }

  Python脚本 (whisper_runner.py):
  #!/usr/bin/env python3
  import whisper
  import json
  import sys
  import argparse

  def main():
      parser = argparse.ArgumentParser()
      parser.add_argument('audio_file')
      parser.add_argument('--model', default='medium.en')
      args = parser.parse_args()

      # 加载模型
      model = whisper.load_model(args.model)

      # 执行转录
      result = model.transcribe(
          args.audio_file,
          fp16=False,
          verbose=False
      )

      # 输出JSON格式
      print(json.dumps({
          'language': result['language'],
          'duration': result.get('duration', 0),
          'text': result['text'],
          'segments': result['segments']
      }))

  if __name__ == '__main__':
      main()
